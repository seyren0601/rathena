//===== rAthena Script =======================================
//= [Job Record] Job Record Book
//===== Description: =========================================
//= Menu of the job record system: view, switch, create and
//= rename job records. Opened with the Job Record Book item
//= (id 900000), which every character receives bound to it
//= when logging in.
//=
//= A job record is one build of the character: job, base and
//= job level, stats, skills, hotkeys, look, equipment and
//= job-quest progress. Only one record is active at a time.
//=
//= See docs/subsystems/map/job-record-system-design.md
//===== Script commands used: ================================
//= getjobrecords;                     fills @jobrecord_* arrays
//= jobrecord_canact()                 1 if an action is possible now
//= jobrecord_preview(<id>)            predicted weight rate (%)
//= jobrecord_preview_new(<doram>)     same, for a new record
//= jobrecord_switch(<id>)             starts a switch
//= jobrecord_create("<name>",<doram>) creates a new record
//= jobrecord_rename(<id>,"<name>")    renames a record
//= On failure the reason is in @jobrecord_error$.
//============================================================

function	script	F_JobRecord	{
	.@npc$ = "[Job Record Book]";

	while (1) {
		if (getjobrecords() < 0) {
			mes .@npc$;
			mes "Your records are still loading, please try again in a moment.";
			close;
		}

		.@max = getbattleflag("job_record_max");

		mes .@npc$;
		if (@jobrecord_active == 0) {
			mes "Current build: ^0055FF" + jobname(Class) + "^000000, Base " + BaseLevel + " / Job " + JobLevel + ".";
			mes "It is not saved as a record yet. It will be saved as \"Default\" the first time you create a record.";
		} else {
			for (.@i = 0; .@i < @jobrecord_count; .@i++) {
				if (@jobrecord_id[.@i] == @jobrecord_active) {
					mes "Active record: ^0055FF" + @jobrecord_name$[.@i] + "^000000";
					mes "(" + jobname(Class) + ", Base " + BaseLevel + " / Job " + JobLevel + ")";
					break;
				}
			}
		}
		mes "Records: " + @jobrecord_count + ((.@max > 0) ? " / " + .@max : "");

		switch (select("Switch to another record:Create a new record:Rename a record:Close")) {
		case 1:
			callsub L_Switch;
			break;
		case 2:
			callsub L_Create;
			break;
		case 3:
			callsub L_Rename;
			break;
		default:
			close;
		}
		next;
	}

// Lets the player pick a record, 10 per page.
// getarg(0): 1 to leave out the active record
// Returns the index in the @jobrecord_* arrays, or -1 when cancelled.
L_Pick:
	.@npc$ = "[Job Record Book]"; // callsub has its own .@ scope
	.@exclude_active = getarg(0);
	deletearray .@list;
	.@total = 0;

	for (.@i = 0; .@i < @jobrecord_count; .@i++) {
		if (.@exclude_active && @jobrecord_id[.@i] == @jobrecord_active)
			continue;
		.@list[.@total] = .@i;
		.@total++;
	}

	if (.@total == 0) {
		next;
		mes .@npc$;
		mes "There is no record to choose.";
		return -1;
	}

	.@page = 0;
	while (1) {
		.@menu$ = "";
		.@shown = 0;

		for (.@i = .@page * 10; .@i < .@total && .@shown < 10; .@i++) {
			.@r = .@list[.@i];
			.@menu$ += replacestr(@jobrecord_name$[.@r], ":", " ") + " - " + jobname(@jobrecord_class[.@r])
				+ " " + @jobrecord_baselv[.@r] + "/" + @jobrecord_joblv[.@r]
				+ ((@jobrecord_id[.@r] == @jobrecord_active) ? " (active)" : "") + ":";
			.@shown++;
		}

		.@has_next = (.@page + 1) * 10 < .@total;
		.@menu$ += (.@has_next ? "Next page" : "") + ":" + (.@page > 0 ? "Previous page" : "") + ":Cancel";

		.@choice = select(.@menu$);

		if (.@choice <= .@shown)
			return .@list[.@page * 10 + .@choice - 1];
		if (.@choice == .@shown + 1)
			.@page++;
		else if (.@choice == .@shown + 2)
			.@page--;
		else
			return -1;
	}

// Shows the weight warning of a preview. getarg(0): predicted weight rate
L_WeightWarning:
	if (getarg(0) >= @jobrecord_warn_rate) {
		mes "^FF0000Warning: your weight will be about " + getarg(0) + "% afterwards.";
		mes "From " + @jobrecord_warn_rate + "% you stop recovering HP and SP naturally.^000000";
	}
	return;

L_Switch:
	.@npc$ = "[Job Record Book]"; // callsub has its own .@ scope
	next;
	mes .@npc$;
	if (!jobrecord_canact()) {
		mes @jobrecord_error$;
		return;
	}
	mes "Which record do you want to switch to?";

	.@r = callsub(L_Pick, 1);
	if (.@r < 0)
		return;

	.@id = @jobrecord_id[.@r];
	.@rate = jobrecord_preview(.@id);

	next;
	mes .@npc$;
	if (.@rate < 0) {
		mes @jobrecord_error$;
		return;
	}
	mes "Switch to ^0055FF" + @jobrecord_name$[.@r] + "^000000?";
	mes "(" + jobname(@jobrecord_class[.@r]) + ", Base " + @jobrecord_baselv[.@r] + " / Job " + @jobrecord_joblv[.@r] + ")";
	mes "Your current equipment stays with your current record, and this record's equipment is put back on.";
	callsub L_WeightWarning, .@rate;

	if (select("Switch:Cancel") == 2)
		return;

	if (!jobrecord_switch(.@id)) {
		next;
		mes .@npc$;
		mes @jobrecord_error$;
		return;
	}

	// The switch finishes once the record has been loaded; the result is shown in the chat window
	close;

L_Create:
	.@npc$ = "[Job Record Book]"; // callsub has its own .@ scope
	next;
	mes .@npc$;
	if (!jobrecord_canact()) {
		mes @jobrecord_error$;
		return;
	}
	mes "Your current build is saved as a record, then you start over at base level 1 with reset stats and skills.";
	mes "Your current equipment stays with your current record.";
	mes "Which race should the new record be?";

	.@race = select("Human (Novice):Doram (Summoner):Cancel");
	if (.@race == 3)
		return;
	.@doram = (.@race == 2);

	next;
	mes .@npc$;
	mes "Enter a name for the new record (1 to 23 characters).";
	mes "For example: Spear Knight.";
	input .@name$;

	.@rate = jobrecord_preview_new(.@doram);

	next;
	mes .@npc$;
	mes "Create ^0055FF" + .@name$ + "^000000 as a " + (.@doram ? "Summoner" : "Novice") + " at base level 1?";
	if (@jobrecord_active == 0)
		mes "Your current build will be saved as the record \"Default\". You can rename it later.";
	if (.@rate >= 0)
		callsub L_WeightWarning, .@rate;

	if (select("Create:Cancel") == 2)
		return;

	next;
	mes .@npc$;
	if (!jobrecord_create(.@name$, .@doram)) {
		mes @jobrecord_error$;
		return;
	}
	mes "Your new record ^0055FF" + .@name$ + "^000000 is ready. Good luck!";
	close;

L_Rename:
	.@npc$ = "[Job Record Book]"; // callsub has its own .@ scope
	next;
	mes .@npc$;
	if (getjobrecords() <= 0) {
		mes "You have no record yet. Create a record first.";
		return;
	}
	mes "Which record do you want to rename?";

	.@r = callsub(L_Pick, 0);
	if (.@r < 0)
		return;

	next;
	mes .@npc$;
	mes "New name for ^0055FF" + @jobrecord_name$[.@r] + "^000000 (1 to 23 characters):";
	input .@name$;

	next;
	mes .@npc$;
	if (!jobrecord_rename(@jobrecord_id[.@r], .@name$)) {
		mes @jobrecord_error$;
		return;
	}
	mes "The record is now called ^0055FF" + .@name$ + "^000000.";
	return;
}

// Gives every character its bound Job Record Book (new characters and existing ones)
-	script	JobRecord_Book	-1,{
OnPCLoginEvent:
	if (countitem(.item_id) == 0)
		getitembound .item_id, 1, Bound_Char;
	end;

OnInit:
	.item_id = 900000; // Job_Record_Book, db/job_record_item_db.yml
	end;
}
